Demo 1: Android → JS

原理：Android 通过 evaluateJavascript("jsFunction(args)", callback) 调用 WebView 中的 JS 函数，可同步获取返回值。
对比旧方案：loadUrl("javascript:...") 无法获取返回值，且 API 18 以下有安全风险。

JS 函数定义

// 1. 显示消息（Android 调用此函数）
function showMessage(msg) {
    document.getElementById('demoArea').textContent = msg;
    return 'JS received: ' + msg;  // 返回值给 Android
}

// 2. 返回 JSON 数据
function getUserData() {
    return JSON.stringify({name:'Peter', age:28, skills:['Android','Kotlin']});
}

// 3. 更新页面内容
function updateContent(text) {
    var el = document.getElementById('demoArea');
    el.textContent = text;  // 使用 textContent 而非 innerHTML
    return 'updated';
}
